

CREATE PROCEDURE [dbo].[SP_New_Dashboard]	
	 @Status varchar(50)='',
	 @User_Type NVARCHAR(30)='',
	 @USER_ID INT=0,
     @Store_code NVARCHAR(20)='',
	 @Store_ID INT =0,
	 @SearchTerm VARCHAR(100) = '',
	 @PageIndex INT =1,
     @PageSize INT = 10,
	 @RecordCount INT=0 OUTPUT,
	 @TotalCount INT=0 OUTPUT,
     @PROCESSED_HU INT=0 OUTPUT,
	 @UNPROCESSED_HU INT=0 OUTPUT,
     @PROCESSED_ARTICLE_QTY INT=0 OUTPUT,
	 @QTY INT=0 OUTPUT,
	 @ENCODED_QTY INT=0 OUTPUT,
     @DIFF_QTY INT=0 OUTPUT,
	 @DPOS_SALE INT=0 OUTPUT,
     @RFID_CHECKOUT INT=0 OUTPUT,
     @RFID_DPOS_SALE INT=0 OUTPUT,
	 @MATCHING_WITH_DPOS_SALE INT=0 OUTPUT,
	 @NOT_MATCHING_WITH_DPOS_SALE INT=0 OUTPUT,
	 @NOT_MATCHING_WITH_RFID_CHECKOUT INT=0 OUTPUT,
	 @TAFFETA_SALE INT=0 OUTPUT,
	 @MANUAL_SALE INT=0 OUTPUT,
	 @DPOS_VOID INT=0 OUTPUT,
	 @RFID_VOID INT=0 OUTPUT,
	 @DIFF_VOID INT=0 OUTPUT,
	 @CURR_DATE DATE= NULL,
	 @PREV_DATE DATE= NULL,
	 @HU_VALIDATED_QTY INT = 0 OUTPUT,
     @HU_WRONG_QTY INT = 0 OUTPUT,
	 @HHT_VALIDATE_QTY INT = 0 OUTPUT,
	 @HU_DIS_ACTUALQTY INT = 0 OUTPUT,
     @HU_DIS_SCANNEDQTY INT = 0 OUTPUT,
	 @HU_DIS_DIFFQTY INT = 0 OUTPUT,
	 @HU_DIFF_TILL_DATE INT = 0 OUTPUT,
	 @SortColumn NVARCHAR(100)='',
     @SortDirection NVARCHAR(10)='',
	 @SortType NVARCHAR(20) = 'string',
	 @SQL NVARCHAR(MAX)='',
	 @fromdate DATE ='',
     @todate DATE =''
	

AS
BEGIN
	SET NOCOUNT ON;

    IF @CURR_DATE IS NULL SET @CURR_DATE = CONVERT(DATE, GETDATE());
	IF @PREV_DATE IS NULL SET @PREV_DATE = CONVERT(DATE, GETDATE()-1);

	DECLARE @Dt DATE = GETDATE()-1;
	DECLARE @FromDt DATETIME = @Dt;
	DECLARE @ToDt   DATETIME = DATEADD(DAY, 1, @Dt);

	--IF @CURR_DATE IS NULL SET @CURR_DATE = CONVERT(DATE, GETDATE()-3);
	--IF @PREV_DATE IS NULL SET @PREV_DATE = CONVERT(DATE, GETDATE()-3);

	--DECLARE @Dt DATE = GETDATE()-3;
	--DECLARE @FromDt DATETIME = @Dt;
	--DECLARE @ToDt   DATETIME = DATEADD(DAY, 3, @Dt);

    SELECT @User_Type=User_Type FROM dbo.[User_Registration] WHERE User_ID=@User_ID;
	SELECT @Store_code=dbo.[tbl_Store_Master].Store_code FROM dbo.[tbl_Store_Master] JOIN dbo.[User_Registration] on dbo.[tbl_Store_Master].Store_ID=dbo.[User_Registration].Store_ID 
	WHERE User_ID=@User_ID;

	DECLARE @StartRow INT = (@PageIndex - 1) * @PageSize + 1;
	DECLARE @EndRow   INT =  @PageIndex * @PageSize;
	DECLARE @OrderBy NVARCHAR(200);

	
	---------------DC dashboard---------------------------------------

	IF (@Status='DC_VALIDATE_DASHBOARD')
	BEGIN
		SELECT FORMAT(@CURR_DATE, 'yyyy-MM-dd hh:mm tt dddd') AS 'DATE',
		FORMAT(DATEADD(DAY,-7, @CURR_DATE), 'yyyy-MM-dd hh:mm tt dddd') AS 'LASTDATE',
		dbo.tbl_Store_Master.Store_Code AS 'Reciving_Plant',dbo.tbl_Store_Master.Store_Name AS 'STORE_NAME',
		COUNT(DISTINCT CASE WHEN dbo.tbl_SAP_DC_Outward_Dtl.Is_Complete_Status_Flg = '1' THEN dbo.tbl_SAP_DC_Outward_Dtl.HU_No END) AS 'PROCESSED_HU',
		COUNT(DISTINCT CASE WHEN dbo.tbl_SAP_DC_Outward_Dtl.Is_Complete_Status_Flg = '0' THEN dbo.tbl_SAP_DC_Outward_Dtl.HU_No END) AS 'UNPROCESSED_HU',
		CAST(SUM(CASE WHEN dbo.tbl_SAP_DC_Outward_Dtl.Is_Complete_Status_Flg = '1' THEN dbo.tbl_SAP_DC_Outward_Dtl.scan_qty ELSE 0 END)AS INT) AS 'PROCESSED_ARTICLE_QTY'
		INTO #DCVALIDATEDASHBOARD
		FROM [dbo].[tbl_store_master] 
		LEFT JOIN dbo.[tbl_SAP_DC_Outward_Dtl] ON dbo.[tbl_store_master].Store_code=dbo.[tbl_SAP_DC_Outward_Dtl].Reciving_Plant
		AND CONVERT (DATE,dbo.[tbl_SAP_DC_Outward_Dtl].Sending_HU_Scan_Dt) = CONVERT(DATE,@CURR_DATE)
		AND dbo.[tbl_SAP_DC_Outward_Dtl].Ref_No IS NOT NULL
		AND dbo.[tbl_SAP_DC_Outward_Dtl].Ref_No <>''
        WHERE (@User_Type IN ('Super Admin','Warehouse Admin') OR (@User_Type = 'Store Admin' AND dbo.tbl_Store_Master.Store_Code = @Store_code))
		AND(@SearchTerm = '' OR dbo.tbl_Store_Master.Store_Code LIKE '%' + @SearchTerm + '%')
		GROUP BY dbo.tbl_Store_Master.Store_Code,dbo.tbl_Store_Master.Store_Name ORDER BY dbo.tbl_Store_Master.Store_Code;

		SET @OrderBy =	
			CASE @SortColumn
				WHEN 'PROCESSED HU QTY' THEN 'R.PROCESSED_HU'
				WHEN 'UNPROCESSED HU QTY' THEN 'R.UNPROCESSED_HU'
				WHEN 'VALIDATED HU ARTICLE QTY' THEN 'R.PROCESSED_ARTICLE_QTY'
				WHEN 'Reciving_Plant' THEN 'R.Reciving_Plant'
				ELSE 'R.Reciving_Plant'
				END + ' ' +
			CASE 
				WHEN @SortDirection = 'desc' THEN 'DESC'
				ELSE 'ASC'
				END;

			SET @SQL = N'SELECT *FROM (SELECT ROW_NUMBER() OVER (ORDER BY ' + @OrderBy + N') AS RowNumber,* FROM #DCVALIDATEDASHBOARD R ) T
			WHERE RowNumber BETWEEN @StartRow AND @EndRow
			ORDER BY RowNumber;';

			EXEC sp_executesql @SQL,
			N'@StartRow INT, @EndRow INT',
			@StartRow = @StartRow,
			@EndRow   = @EndRow;

       SELECT @PROCESSED_HU = ISNULL(SUM(PROCESSED_HU),0),@UNPROCESSED_HU = ISNULL(SUM(UNPROCESSED_HU),0),
	   @PROCESSED_ARTICLE_QTY = ISNULL(SUM(PROCESSED_ARTICLE_QTY),0), @RecordCount=COUNT(*) FROM #DCVALIDATEDASHBOARD;

	-- Clean up the temporary table
	   DROP TABLE #DCVALIDATEDASHBOARD;

    END 

	---------RETURN DASHBOARD--------------------------

	ELSE IF (@Status='RETURN_DASHBOARD')
	BEGIN
		SELECT FORMAT(@PREV_DATE, 'yyyy-MM-dd hh:mm tt dddd') AS 'DATE',
		FORMAT(DATEADD(DAY,-7, @PREV_DATE), 'yyyy-MM-dd hh:mm tt dddd') AS 'LASTDATE',
		V.Store_Code,V.Store_Name AS 'STORE_NAME',SUM(V.RET_QTY) AS 'RETURN_QTY',
		SUM(CASE WHEN V.RET_QTY < ISNULL(E.ENCODE_QTY,0)THEN V.RET_QTY ELSE ISNULL(E.ENCODE_QTY,0)END) AS'ENCODE_QTY',
		SUM(V.RET_QTY)-SUM(CASE WHEN V.RET_QTY < ISNULL(E.ENCODE_QTY,0)THEN V.RET_QTY ELSE ISNULL(E.ENCODE_QTY,0)END) AS 'DIFFERENCE_QTY'
		INTO #RETURNDASHBOARD
		FROM
		(SELECT [dbo].[tbl_Store_Master].Store_Code,[dbo].[tbl_Store_Master].Store_Name,[dbo].[VPOS_SALE_RET_DTL].SCAN_ITEM_EAN_CD,CAST(ISNULL(RET_QTY,0) AS INT) AS 'RET_QTY'
		 FROM [dbo].[tbl_Store_Master] LEFT JOIN [dbo].[VPOS_SALE_RET_DTL] ON [dbo].[tbl_Store_Master].Store_Code=[dbo].[VPOS_SALE_RET_DTL].STORE_CODE
		 AND [dbo].[VPOS_SALE_RET_DTL].BILL_DATE=CONVERT(DATE,@PREV_DATE)
		 AND SUBSTRING (dbo.[VPOS_SALE_RET_DTL].item_cd,1,3) in ('111','112','113') AND [dbo].[VPOS_SALE_RET_DTL].COUNTER_NO IN ('RO','RE')
		 )AS  V
		LEFT JOIN 
		(SELECT [dbo].[tbl_Encoding_Dtl].EAN,[dbo].[tbl_Store_Master].Store_Code,COUNT([dbo].[tbl_Encoding_Dtl].Encode_Trans_ID) AS 'ENCODE_QTY'
		FROM [dbo].[tbl_Encoding_Dtl] INNER JOIN [dbo].[tbl_Store_Master] ON [dbo].[tbl_Encoding_Dtl].Store_ID = [dbo].[tbl_Store_Master].Store_ID
		WHERE CONVERT(DATE,[dbo].[tbl_Encoding_Dtl].encode_date)>= CONVERT (DATE,@PREV_DATE)
		AND [dbo].[tbl_Encoding_Dtl].STORE_FLOOR_ID IS NOT NULL AND [dbo].[tbl_Encoding_Dtl].STORE_ID IS NOT NULL AND [dbo].[tbl_Encoding_Dtl].WH_ID IS NULL
		GROUP BY[dbo].[tbl_Encoding_Dtl].EAN,[dbo].[tbl_Store_Master].Store_Code
		) AS E ON V.Store_Code = E.Store_Code AND V.SCAN_ITEM_EAN_CD = E.EAN 
		WHERE (@SearchTerm = '' OR V.Store_Code LIKE '%' + @SearchTerm + '%')
		GROUP BY V.Store_Code,V.Store_Name  
		HAVING (@User_Type = 'Super Admin' OR (@User_Type = 'Store Admin' AND V.Store_Code = @Store_code)) ORDER BY V.Store_Code

		SELECT @QTY = ISNULL(SUM(RETURN_QTY),0),@ENCODED_QTY = ISNULL(SUM(ENCODE_QTY),0),@DIFF_QTY = ISNULL(SUM(DIFFERENCE_QTY),0),
		@RecordCount=COUNT(*) FROM #RETURNDASHBOARD;

		SET @SortDirection =
				CASE 
					WHEN LOWER(@SortDirection) = 'desc' THEN 'DESC'
					ELSE 'ASC'
				END;

			SET @OrderBy =	
				CASE @SortColumn
					WHEN 'RETURN QTY' THEN 'R.RETURN_QTY' WHEN 'ENCODED VS RETURN (QTY)' THEN 'R.ENCODE_QTY'
					WHEN 'PENDING QTY' THEN 'R.DIFFERENCE_QTY' WHEN 'STORE' THEN 'R.Store_Code'
					ELSE 'R.Store_Code'
				END
				+ ' ' + @SortDirection
				+ ', R.Store_Code ASC';

           SET @SQL = N' SELECT * FROM (SELECT	ROW_NUMBER() OVER (ORDER BY ' + @OrderBy + N') AS RowNumber,*
					FROM #RETURNDASHBOARD R ) T
					WHERE RowNumber BETWEEN @StartRow AND @EndRow
					ORDER BY RowNumber;';

			EXEC sp_executesql
			@SQL,N'@StartRow INT, @EndRow INT',
			@StartRow = @StartRow,	@EndRow   = @EndRow;
	
    -- Clean up the temporary table
	   DROP TABLE #RETURNDASHBOARD;
    END

	------------VOID DASHBOARD---------------------------------

	ELSE IF (@Status='VOID_DASHBOARD')
	BEGIN
          WITH BadEPC AS (SELECT EPC_TAG FROM dbo.VPOS_SCAN_ITEMS_DTL GROUP BY EPC_TAG HAVING COUNT(CASE WHEN EXT_COL = 'V' THEN 1 END) > 0
					 AND COUNT(CASE WHEN EXT_COL IS NULL THEN 1 END) > 0),

		  VoidBase AS (SELECT V.STORE_CODE,V.EPC_TAG,V.SCAN_ITEM_EAN_CD,ROW_NUMBER() OVER (PARTITION BY V.EPC_TAG, V.STORE_CODE ORDER BY V.LAST_UPDATED_DATETIME DESC) AS rn FROM dbo.VPOS_SCAN_ITEMS_DTL V
					   WHERE V.LAST_UPDATED_DATETIME >= @PREV_DATE AND V.LAST_UPDATED_DATETIME < DATEADD(DAY, 1, @PREV_DATE)
					   AND (V.item_cd LIKE '111%' OR V.item_cd LIKE '112%' OR V.item_cd LIKE '113%')
					   AND V.EXT_COL = 'V' AND NOT EXISTS (SELECT 1 FROM BadEPC B WHERE B.EPC_TAG = V.EPC_TAG)),

		  VoidCTE AS (SELECT STORE_CODE,SCAN_ITEM_EAN_CD,COUNT(EPC_TAG) AS VOID_QTY FROM VoidBase WHERE rn = 1 GROUP BY STORE_CODE,SCAN_ITEM_EAN_CD),

		  EncodeCTE AS (SELECT SM.Store_Code,E.EAN,COUNT(E.Encode_Trans_ID) AS ENCODE_QTY FROM dbo.tbl_Encoding_Dtl E JOIN dbo.tbl_Store_Master SM ON E.Store_ID = SM.Store_ID
					   WHERE E.encode_date >= @PREV_DATE AND E.encode_date < GETDATE()
					   AND E.STORE_FLOOR_ID IS NOT NULL AND E.STORE_ID IS NOT NULL AND E.WH_ID IS NULL GROUP BY SM.Store_Code, E.EAN)

		 SELECT FORMAT(@PREV_DATE, 'yyyy-MM-dd hh:mm tt dddd') AS 'DATE',
		 FORMAT(DATEADD(DAY,-7, @PREV_DATE), 'yyyy-MM-dd hh:mm tt dddd') AS 'LASTDATE',
		 SM.Store_Code AS 'STORE',SM.Store_Name AS 'STORE_NAME',ISNULL(SUM(V.VOID_QTY), 0) AS 'VOID_QTY',
		 ISNULL(SUM(CASE WHEN V.VOID_QTY < ISNULL(E.ENCODE_QTY, 0) THEN V.VOID_QTY ELSE ISNULL(E.ENCODE_QTY, 0)END), 0) AS 'ENCODE_QTY',
		 ISNULL(SUM(V.VOID_QTY), 0) - ISNULL(SUM(CASE WHEN V.VOID_QTY < ISNULL(E.ENCODE_QTY, 0) THEN V.VOID_QTY ELSE ISNULL(E.ENCODE_QTY, 0) END), 0) AS 'DIFFERENCE_QTY'
		 INTO #VOIDDASHBOARD
		 FROM dbo.tbl_Store_Master SM 
		 LEFT JOIN VoidCTE V  ON SM.Store_Code = V.STORE_CODE
		 LEFT JOIN EncodeCTE E ON SM.Store_Code = E.Store_Code AND V.SCAN_ITEM_EAN_CD = E.EAN
		 WHERE (@SearchTerm = '' OR SM.Store_Code LIKE '%' + @SearchTerm + '%')
		 GROUP BY SM.Store_Code,SM.Store_Name HAVING (@User_Type = 'Super Admin' OR (@User_Type = 'Store Admin' AND SM.Store_Code = @Store_code)) ORDER BY SM.Store_Code;

		SELECT @QTY = ISNULL(SUM(VOID_QTY),0),@ENCODED_QTY = ISNULL(SUM(ENCODE_QTY),0),
		@DIFF_QTY = ISNULL(SUM(DIFFERENCE_QTY),0),@RecordCount=COUNT(*) FROM #VOIDDASHBOARD;

			SET @OrderBy =	
			CASE @SortColumn
				WHEN 'VOID QTY' THEN 'R.VOID_QTY' WHEN 'ENCODED VS VOID (QTY)' THEN 'R.ENCODE_QTY'
				WHEN 'PENDING QTY' THEN 'R.DIFFERENCE_QTY' WHEN 'STORE'THEN 'R.STORE'
				ELSE 'R.STORE'
				END + ' ' +
			CASE 
				WHEN @SortDirection = 'desc' THEN 'DESC'
				ELSE 'ASC'
				END;

			SET @SQL = N'
			SELECT * FROM (SELECT ROW_NUMBER() OVER (ORDER BY ' + @OrderBy + N') AS RowNumber,* FROM #VOIDDASHBOARD R ) T
			WHERE RowNumber BETWEEN @StartRow AND @EndRow
			ORDER BY RowNumber;';

			EXEC sp_executesql
			@SQL,
			N'@StartRow INT, @EndRow INT',
			@StartRow = @StartRow,
			@EndRow   = @EndRow;
	
     -- Clean up the temporary table
		DROP TABLE #VOIDDASHBOARD;
    END

	-----------GRC STORE DASHBOARD--------------------------------------------
    ELSE IF (@Status='STORE_DASHBOARD')
	BEGIN
		WITH  STOREBASE AS (SELECT DISTINCT S.Store_Code as STORE,S.Store_Name AS 'STORE_NAME' FROM dbo.[tbl_Store_Master] S ),

        LAST_GRC_DATE AS (SELECT STORE_CODE,MAX(TRY_CONVERT(DATE, GRC_DATE, 105)) AS LAST_GRC_DATE,DATEADD(DAY, -7, MAX(TRY_CONVERT(DATE, GRC_DATE, 105))) AS LAST7_GRC_DATE FROM dbo.tbl_GRC_DETAILS GROUP BY STORE_CODE),

        GRCBASE AS (SELECT g.Store_Code AS STORE,COUNT(DISTINCT g.HU) AS 'HU_RECEIVED_QTY',
		     COUNT(DISTINCT CASE WHEN g.Is_Status = 1 THEN g.HU END) AS 'HU_VALIDATED_QTY',
			 --COUNT(DISTINCT CASE WHEN g.Is_Status = 0 THEN g.HU END) AS 'HU_WRONG_QTY',
			 lg.LAST_GRC_DATE,lg.LAST7_GRC_DATE
			 FROM dbo.tbl_GRC_DETAILS g JOIN LAST_GRC_DATE lg  ON g.STORE_CODE = lg.STORE_CODE
			 AND TRY_CONVERT(DATE, g.GRC_DATE, 105) = lg.LAST_GRC_DATE GROUP BY g.STORE_CODE,lg.LAST_GRC_DATE,lg.LAST7_GRC_DATE),

       STOREGRCBASE AS (SELECT sgm.Reciving_Plant AS STORE,COUNT(DISTINCT CASE WHEN sgm.Store_GRC_Complete_Status_Flg = 1 THEN sgm.STORE_GRC_HUNO END) AS 'HHT_VALIDATE',
	                    COUNT(DISTINCT CASE WHEN sgm.Store_GRC_Complete_Status_Flg = 2 THEN sgm.STORE_GRC_HUNO END) AS 'HU_WRONG_QTY'
		     FROM dbo.tbl_STORE_GRC_MST_DTL sgm  
			 INNER JOIN dbo.tbl_GRC_DETAILS g ON g.HU = sgm.STORE_GRC_HUNO AND g.STORE_CODE = sgm.Reciving_Plant
			 INNER JOIN LAST_GRC_DATE lg ON g.STORE_CODE = lg.STORE_CODE AND TRY_CONVERT(DATE, g.GRC_DATE, 105) = lg.LAST_GRC_DATE
			 GROUP BY sgm.Reciving_Plant)

       SELECT FORMAT(gb.LAST_GRC_DATE, 'yyyy-MM-dd hh:mm tt dddd') AS 'DATE',
	   FORMAT(gb.LAST7_GRC_DATE, 'yyyy-MM-dd hh:mm tt dddd') AS 'LASTDATE',
	   SB.STORE,SB.STORE_NAME,
	   ISNULL(GB.HU_RECEIVED_QTY,0) AS 'HU_RECEIVED_QTY', 
	   ISNULL(GB.HU_VALIDATED_QTY, 0) AS 'HU_VALIDATED_QTY',
	   ISNULL(SGB.HHT_VALIDATE, 0) AS 'HHT_VALIDATE_QTY',
	   ISNULL(GB.HU_RECEIVED_QTY,0) - (ISNULL(SGB.HHT_VALIDATE, 0) + ISNULL(SGB.HU_WRONG_QTY, 0)) AS 'STORE_PENDING_QTY',
	   ISNULL(SGB.HU_WRONG_QTY, 0) AS 'HU_WRONG_QTY'
	   INTO #STOREDASHBOARD
       FROM STOREBASE SB LEFT JOIN  GRCBASE GB ON  SB.STORE = GB.STORE  LEFT JOIN STOREGRCBASE SGB ON SB.STORE=SGB.STORE
	   WHERE (@User_Type = 'Super Admin' OR (@User_Type = 'Store Admin' AND SB.STORE = @Store_code))
	   AND (@SearchTerm = '' OR SB.STORE LIKE '%' + @SearchTerm + '%');


		SELECT @QTY = ISNULL(SUM(HU_RECEIVED_QTY),0),
		@HU_VALIDATED_QTY = ISNULL(SUM(HU_VALIDATED_QTY),0), @HHT_VALIDATE_QTY   = ISNULL(SUM(HHT_VALIDATE_QTY), 0), @HU_WRONG_QTY   = ISNULL(SUM(HU_WRONG_QTY), 0),
		@ENCODED_QTY = ISNULL(SUM(STORE_PENDING_QTY),0),
		@RecordCount=COUNT(*) FROM #STOREDASHBOARD;

		SET @OrderBy =	
			CASE @SortColumn
			    WHEN 'HU RECEIVED QTY' THEN 'R.HU_RECEIVED_QTY'
				WHEN 'WH VALIDATED QTY' THEN 'R.HU_VALIDATED_QTY' 
				WHEN 'WRONG HU QTY' THEN 'R.HU_WRONG_QTY' 
				WHEN 'STORE VALIDATED QTY' THEN 'R.HHT_VALIDATE_QTY' 
				WHEN 'STORE PENDING FOR VALIDATION (QTY)' THEN 'R.STORE_PENDING_QTY' 
				WHEN 'STORE'THEN 'R.STORE'
				ELSE 'R.STORE'
				END + ' ' +
			CASE 
				WHEN @SortDirection = 'desc' THEN 'DESC'
				ELSE 'ASC'
				END;

			SET @SQL = N'
			SELECT * FROM (SELECT ROW_NUMBER() OVER (ORDER BY ' + @OrderBy + N') AS RowNumber,* FROM #STOREDASHBOARD R ) T
			WHERE RowNumber BETWEEN @StartRow AND @EndRow
			ORDER BY RowNumber;';

			EXEC sp_executesql
			@SQL,
			N'@StartRow INT, @EndRow INT',
			@StartRow = @StartRow,
			@EndRow   = @EndRow;
	

	-- Clean up the temporary table
		DROP TABLE #STOREDASHBOARD;
    END


  ----------STOCK DASHBOARD---------

	ELSE IF (@Status='STORE_STOCK_DASHBOARD')
	BEGIN
		SELECT FORMAT(@CURR_DATE, 'yyyy-MM-dd hh:mm tt dddd') AS 'DATE',
		S.Store_Code  AS STORE,S.Store_Name AS 'STORE_NAME',COUNT(DISTINCT E.Encode_EPC) AS 'RFID_STOCK' 
		INTO #STORESTOCKDASHBOARD
		FROM dbo.[tbl_Store_Master] S 
		LEFT JOIN dbo.[tbl_Encoding_Dtl] E ON S.Store_ID=E.Store_ID AND E.Is_Status IN (1,2)
		WHERE (@User_Type = 'Super Admin' OR (@User_Type = 'Store Admin' AND S.Store_Code = @Store_code))
		AND (@SearchTerm = '' OR S.Store_Code LIKE '%' + @SearchTerm + '%')
		GROUP BY S.Store_Code,S.Store_Name ORDER BY S.Store_Code;

		SELECT @ENCODED_QTY = ISNULL(SUM(RFID_STOCK),0), @RecordCount=COUNT(*) FROM #STORESTOCKDASHBOARD;
	
	-- Retrieve the paginated results
		SET @OrderBy =	
			CASE @SortColumn
				WHEN 'RFID STOCK' THEN 'R.RFID_STOCK' WHEN 'STORE' THEN 'R.STORE'
				ELSE 'R.STORE'
				END + ' ' +
			CASE 
				WHEN @SortDirection = 'desc' THEN 'DESC'
				ELSE 'ASC'
				END;

			SET @SQL = N'SELECT * FROM (SELECT	ROW_NUMBER() OVER (ORDER BY ' + @OrderBy + N') AS RowNumber,* FROM #STORESTOCKDASHBOARD R ) T
			WHERE RowNumber BETWEEN @StartRow AND @EndRow
			ORDER BY RowNumber;';

			EXEC sp_executesql
			@SQL,
			N'@StartRow INT, @EndRow INT',
			@StartRow = @StartRow,
			@EndRow   = @EndRow;


		SELECT COUNT(DISTINCT TID) AS 'FRESH_RFID_TAG' FROM dbo.[tbl_Encoding_Dtl] WHERE Is_Status IN (3,4) --AND CONVERT(DATE,Encode_Date) = CONVERT(DATE, GETDATE()-2)
		AND (@User_Type = 'Super Admin' OR (@User_Type = 'Store Admin' AND Store_ID = @Store_ID)) 


	-- Clean up the temporary table
		DROP TABLE #STORESTOCKDASHBOARD;
	END


	---------SALE DASHBOARD-----------------------------

	ELSE IF(@Status='SALE_DASHBOARD')
	BEGIN
		
		WITH POS_BASE AS (
		SELECT S.STORE_CODE,ISNULL(SUM(SALE_QTY),0) AS Total_POS_Sale FROM TBL_STORE_MASTER S LEFT JOIN VPOS_SALE_RET_DTL V ON V.STORE_CODE=S.Store_CODE
		AND BILL_DATE = @Dt AND COUNTER_NO NOT IN ('RO', 'RE') AND LEFT(ITEM_CD, 3) IN ('111','112','113')
		GROUP BY S.STORE_CODE
					  ),

		RFID_CHECKOUT AS (
		SELECT CHECKOUT_STORE_CODE AS STORE_CODE,COUNT(DISTINCT Encode_EPC) AS Total_RFID_Checkout--,COUNT(DISTINCT CASE WHEN Void_Flg = 'V' THEN Encode_EPC END) AS RFID_Void
		FROM tbl_Encoding_Dtl WHERE Is_Status IN ('3','4') AND Checkout_DtTime >= @FromDt AND Checkout_DtTime <  @ToDt
		GROUP BY CHECKOUT_STORE_CODE
					  ),

		RFID_POS AS (
		SELECT STORE_CODE,COUNT(DISTINCT EPC_TAG) AS Total_RFID_POS_Sale,COUNT(DISTINCT CASE WHEN MANUAL_SCAN_FLAG = '1' THEN TRANS_ID END) AS Total_Manual_Sale
		--,COUNT(DISTINCT CASE WHEN EXT_COL = 'V' THEN TRANS_ID END) AS Total_Void 
		FROM VPOS_SCAN_ITEMS_DTL
		WHERE Is_Status = '3'AND CHECKOUT_DATETIME >= @FromDt AND CHECKOUT_DATETIME <  @ToDt GROUP BY STORE_CODE
				 ),

        VOIDS AS (
		SELECT BB.STORE_CODE,COUNT(DISTINCT BB.EPC_TAG) AS Total_Void,COUNT(DISTINCT CASE WHEN Void_Sync_Flg  = '1' THEN BB.EPC_TAG END) AS RFID_Void FROM (
        SELECT V.EPC_TAG,V.STORE_CODE,V.Void_Sync_Flg,ROW_NUMBER() OVER (PARTITION BY V.EPC_TAG, V.STORE_CODE ORDER BY V.LAST_UPDATED_DATETIME DESC) AS rn
        FROM VPOS_SCAN_ITEMS_DTL V WHERE V.EXT_COL = 'V' AND V.LAST_UPDATED_DATETIME >= @FromDt AND V.LAST_UPDATED_DATETIME < @ToDt AND (V.item_cd LIKE '111%' OR V.item_cd LIKE '112%' OR V.item_cd LIKE '113%')
        AND NOT EXISTS
        (SELECT 1 FROM VPOS_SCAN_ITEMS_DTL VP WHERE VP.EPC_TAG = V.EPC_TAG GROUP BY VP.EPC_TAG HAVING COUNT(CASE WHEN VP.EXT_COL = 'V' THEN 1 END) > 0 
         AND COUNT(CASE WHEN VP.EXT_COL IS NULL THEN 1 END) > 0)
         )BB WHERE BB.RN=1 GROUP BY BB.STORE_CODE),

		MATCHING AS (
		SELECT s.STORE_CODE,COUNT(DISTINCT e.Encode_EPC) AS Checkout_Matching_POS,
        COUNT(DISTINCT CASE WHEN e.Is_Status not IN ('3','4') THEN e.Encode_EPC END) AS RFID_Not_Matching_POS,
        COUNT(DISTINCT CASE WHEN e.Encode_EPC IS NULL OR e.Is_Status NOT IN ('3','4')THEN s.EPC_TAG END) AS POS_Not_Matching_RFID

		INTO #SALEDASHBOARD

		LEFT JOIN MATCHING m ON p.STORE_CODE = m.STORE_CODE
		WHERE (@User_Type = 'Super Admin' OR (@User_Type = 'Store Admin' AND p.Store_Code = @Store_code))
		AND(@SearchTerm = '' OR P.Store_Code LIKE '%' + @SearchTerm + '%')
		ORDER BY SM.STORE_CODE,SM.Store_Name;

			WHEN 'TOTAL RFID SALE' THEN 'R.TOTAL_RFID_DPOS_SALE'

			WHEN 'RFID CHECKOUT NOT MATCHING WITH DPOS SALE' THEN 'R.RFID_CHECKOUT_NOT_MATCHING_WITH_DPOS_SALE'
			WHEN 'DPOS SALE NOT MATCHING WITH RFID CHECKOUT' THEN 'R.DPOS_SALE_NOT_MATCHING_WITH_RFID_CHECKOUT'

			WHEN 'TOTAL DIFF VOID' THEN 'R.TOTAL_DIFF_VOID'

			ELSE 'ASC'

	BEGIN
       WITH BadEPC AS (SELECT EPC_TAG FROM dbo.VPOS_SCAN_ITEMS_DTL WHERE (LAST_UPDATED_DATETIME >= @fromdate) AND (LAST_UPDATED_DATETIME < DATEADD(DAY, 1, @Todate))
		                  GROUP BY EPC_TAG HAVING COUNT(CASE WHEN EXT_COL = 'V' THEN 1 END) > 0 AND COUNT(CASE WHEN EXT_COL IS NULL THEN 1 END) > 0),

		  VoidBase AS (SELECT V.STORE_CODE,V.EPC_TAG,V.SCAN_ITEM_EAN_CD,CAST(V.LAST_UPDATED_DATETIME AS DATE) AS 'VOID_DATE',ROW_NUMBER() OVER (PARTITION BY V.EPC_TAG, V.STORE_CODE ORDER BY V.LAST_UPDATED_DATETIME DESC) AS rn FROM dbo.VPOS_SCAN_ITEMS_DTL V
					   WHERE V.LAST_UPDATED_DATETIME >= @fromdate AND V.LAST_UPDATED_DATETIME < DATEADD(DAY, 1,@todate)
					   AND (V.item_cd LIKE '111%' OR V.item_cd LIKE '112%' OR V.item_cd LIKE '113%')
					   AND V.EXT_COL = 'V' AND NOT EXISTS (SELECT 1 FROM BadEPC B WHERE B.EPC_TAG = V.EPC_TAG)),

		  VoidCTE AS (SELECT  VOID_DATE,STORE_CODE,SCAN_ITEM_EAN_CD,COUNT(EPC_TAG) AS VOID_QTY FROM VoidBase WHERE rn = 1 GROUP BY STORE_CODE,SCAN_ITEM_EAN_CD,VOID_DATE),

		  EncodeCTE AS (SELECT CAST(E.Encode_Date AS DATE) AS 'ENCODE_DATE',SM.Store_Code,E.EAN,COUNT(E.Encode_Trans_ID) AS ENCODE_QTY FROM dbo.tbl_Encoding_Dtl E JOIN dbo.tbl_Store_Master SM ON E.Store_ID = SM.Store_ID
					   WHERE E.encode_date >= @fromdate AND E.encode_date < DATEADD(DAY, 1, @todate)
					   AND E.STORE_FLOOR_ID IS NOT NULL AND E.STORE_ID IS NOT NULL AND E.WH_ID IS NULL GROUP BY SM.Store_Code, E.EAN,CAST(E.Encode_Date AS DATE))

		 SELECT V.VOID_DATE AS 'DATE',SM.Store_Code AS 'STORE',SM.Store_Name AS 'STORE_NAME',ISNULL(SUM(V.VOID_QTY), 0) AS 'VOID_QTY',
		 ISNULL(SUM(CASE WHEN V.VOID_QTY < ISNULL(E.ENCODE_QTY, 0) THEN V.VOID_QTY ELSE ISNULL(E.ENCODE_QTY, 0)END), 0) AS 'ENCODE_QTY',
		 ISNULL(SUM(V.VOID_QTY), 0) - ISNULL(SUM(CASE WHEN V.VOID_QTY < ISNULL(E.ENCODE_QTY, 0) THEN V.VOID_QTY ELSE ISNULL(E.ENCODE_QTY, 0) END), 0) AS 'DIFFERENCE_QTY'
		 INTO #LAST7DAYVOIDDASHBOARD
		 FROM dbo.tbl_Store_Master SM 
		 LEFT JOIN VoidCTE V  ON SM.Store_Code = V.STORE_CODE
		 LEFT JOIN EncodeCTE E ON SM.Store_Code = E.Store_Code AND V.SCAN_ITEM_EAN_CD = E.EAN AND V.VOID_DATE=E.ENCODE_DATE
		 WHERE (@SearchTerm = '' OR SM.Store_Code LIKE '%' + @SearchTerm + '%')
		 GROUP BY SM.Store_Code,SM.Store_Name,V.VOID_DATE HAVING (@Store_code='' OR SM.Store_Code = @Store_code) AND ISNULL(SUM(V.VOID_QTY), 0) > 0  ORDER BY SM.Store_Code;

		SELECT @QTY = ISNULL(SUM(VOID_QTY),0),@ENCODED_QTY = ISNULL(SUM(ENCODE_QTY),0),
		@DIFF_QTY = ISNULL(SUM(DIFFERENCE_QTY),0),@RecordCount=COUNT(*) FROM #LAST7DAYVOIDDASHBOARD;

			SET @OrderBy =	
			CASE @SortColumn
			    WHEN 'DATE' THEN 'R.DATE'
				WHEN 'VOID QTY' THEN 'R.VOID_QTY' 
				WHEN 'ENCODED VS VOID (QTY)' THEN 'R.ENCODE_QTY'
				WHEN 'PENDING QTY' THEN 'R.DIFFERENCE_QTY' 
				WHEN 'STORE'THEN 'R.STORE'
				WHEN 'STORE_NAME' THEN 'R.STORE_NAME'
				ELSE 'R.DATE'
				END + ' ' +
			CASE 
				WHEN @SortDirection = 'desc' THEN 'DESC'
				ELSE 'ASC'
				END;

			SET @SQL = N'
			SELECT * FROM (SELECT ROW_NUMBER() OVER (ORDER BY ' + @OrderBy + N') AS RowNumber,* FROM #LAST7DAYVOIDDASHBOARD R ) T
			WHERE RowNumber BETWEEN @StartRow AND @EndRow
			ORDER BY RowNumber;';

			EXEC sp_executesql
			@SQL,
			N'@StartRow INT, @EndRow INT',
			@StartRow = @StartRow,
			@EndRow   = @EndRow;
